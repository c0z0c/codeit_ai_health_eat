# 알약 분류 시스템 아키텍처 설계

## 1. 프로젝트 개요

### 1.1. 시스템 목적
PyQt5 기반 데스크톱 애플리케이션에서 촬영한 알약 이미지를 AI 모델로 분석하여 약물 정보 및 병용금기 정보를 제공하는 헬스케어 시스템

### 1.2. 핵심 기능
- **Phase 1**: 알약 객체 탐지 및 분류 (Faster R-CNN)
- **Phase 2 (선택)**: 2단계 파이프라인 (YOLO + EfficientNet)
- **Phase 3 (선택)**: 약물 상호작용 및 위험성 분석

## 5. AI 엔진 아키텍처

### 5.1. PillAnalysisEngine 클래스 구조

```python
class PillAnalysisEngine:
    def __init__(self, model_1_stage_path, model_2_stage_path=None):
        self.device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
        self.database = self.init_database()  # DataFrame 로드
        self.model_1_stage = self.load_1_stage_model()
        self.model_2_stage = self.load_2_stage_model() if model_2_stage_path else None
    
    def analyze_image(self, image) -> dict:
        """메인 분석 파이프라인"""
        validated_image = self.validate_image(image)
        detections = self.detect_pills_fasterrcnn(validated_image)
        classifications = self.classify_pills(validated_image, detections)
        drug_info = self.validate_and_enrich(classifications)
        return self.format_response(drug_info)
```

## 8. 핵심 알고리즘

### 8.1. 병용금기 검색 알고리즘

```python
def find_ddi(self, category_ids):
    """카테고리 ID 리스트로 병용금기 조합 찾기"""
    df_drug = self.database['df_drug']
    df_interaction = self.database['td_interaction']
    
    # 1. 카테고리 → 제품 코드 매핑
    df_filtered = df_drug[df_drug['category_id'].isin(category_ids)]
    codes = df_filtered['code'].tolist()
    
    # 2. 제품 코드 조합으로 병용금기 검색
    df_ddi = pd.DataFrame()
    for code_a in codes:
        for code_b in codes:
            if code_a == code_b: continue
            matched = df_interaction[
                (df_interaction['제품코드A'] == code_a) &
                (df_interaction['제품코드B'] == code_b)
            ]
            if not matched.empty:
                df_ddi = pd.concat([df_ddi, matched])
    
    # 3. 중복 제거 및 반환
    return df_ddi.drop_duplicates().to_dict(orient='records')
```

## 9. 성능 최적화

### 9.1. 모델 다운로드 자동화

```python
def download_model_files(url, target):
    """모델 파일 자동 다운로드 (진행률 표시)"""
    if os.path.exists(target):
        return True
    
    def progress_hook(block_num, block_size, total_size):
        percent = min(100, (block_num * block_size * 100) // total_size)
        sys.stdout.write(f'\r진행률: {percent}%')
    
    urllib.request.urlretrieve(url, target, reporthook=progress_hook)
```

### 9.2. 분할 아카이브 병합

```python
def extract_model_from_split_files():
    """best.tar.001, .002, .003 → best.pt 추출"""
    split_files = glob.glob("best.tar.*")
    split_files.sort()
    
    # 병합
    with open("best.tar", 'wb') as merged:
        for split_file in split_files:
            with open(split_file, 'rb') as f:
                merged.write(f.read())
    
    # 압축 해제
    with tarfile.open("best.tar", 'r') as tar:
        tar.extract('best.pt')
```
